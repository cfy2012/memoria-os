    /* GUI 全量重绘（对齐固件每帧全量重放：面板/列表/按钮/文本/彩蛋） */
    void render_gui() {
        /* 列表多实例 */
        for (auto& lp : lists_) {
            if (!lp.on) continue;
            int gw = lp.w / 5, gh = lp.h / 10; if (gw < 1) gw = 1; if (gh < 1) gh = 1;
            canvas.fill(lp.x, lp.y, lp.w, lp.h, ' ');
            canvas.rect(lp.x, lp.y, lp.w, lp.h, '+');
            std::vector<std::string> rows;
            for (auto& m : lp.lines) {
                std::string s = m;
                if (s.empty()) { rows.push_back(""); continue; }
                while ((int)s.size() > gw - 2) { rows.push_back(s.substr(0, gw - 2)); s = s.substr(gw - 2); }
                rows.push_back(s);
            }
            int avail = gh - 2;
            int skip = (int)rows.size() - avail - lp.scroll * avail;
            if (skip < 0) skip = 0;
            for (int j = 0; j < avail && skip + j < (int)rows.size(); j++)
                canvas.put_text(lp.x + 5, lp.y + 10 + j * 10, rows[skip + j]);
        }
        /* 按钮：指定位置画框 + 文字；聚焦 > 标记；font 放大字符格 */
        for (int i = 0; i < (int)btns_.size(); i++) {
            auto& b = btns_[i];
            int gw = b.w / 5, gh = b.h / 10; if (gw < 1) gw = 1; if (gh < 1) gh = 1;
            canvas.fill(b.x, b.y, b.w, b.h, ' ');
            canvas.rect(b.x, b.y, b.w, b.h, (i == bsel_i) ? '>' : '+');
            int f = b.font < 1 ? 1 : (b.font > 3 ? 3 : b.font);
            if (!b.label.empty()) {
                int gx = b.x / 5, gy = b.y / 10;
                size_t maxc = (size_t)std::max(0, gw - 2);
                int off = (gw - (int)(b.label.size() * f)) / 2; if (off < 0) off = 0;
                for (size_t k = 0; k < b.label.size() && gx + off + (int)k * f < CW; k++) {
                    for (int r = 0; r < f; r++)
                        if (gx + off + (int)k * f + r < CW)
                            canvas.cells[gy + gh / 2][gx + off + (int)k * f + r].ch = b.label[k];
                }
            }
        }
        /* 文本控件 lbl */
        for (auto& l : lbls_) {
            if (l.text.empty()) continue;
            int f = l.font < 1 ? 1 : (l.font > 3 ? 3 : l.font);
            int gx = l.x / 5, gy = l.y / 10;
            for (size_t k = 0; k < l.text.size() && gx + (int)k * f < CW; k++) {
                for (int r = 0; r < f; r++)
                    if (gx + (int)k * f + r < CW)
                        canvas.cells[gy][gx + (int)k * f + r].ch = l.text[k];
            }
        }
        /* 彩蛋卡 */
        if (egg_card_on) {
            int gx = egg_x / 5, gy = egg_y / 10;
            canvas.rect(egg_x, egg_y, 340, 150, '+');
            std::string head = std::string(EGG_NO[egg_idx]) + "  " + EGG_TITLE[egg_idx];
            canvas.put_text(egg_x + 5, egg_y + 10, head);
            canvas.put_text(egg_x + 10, egg_y + 50, EGG_HINT[egg_idx]);
            canvas.put_text(egg_x + 10, egg_y + 90, std::string("章节: ") + EGG_WHERE[egg_idx]);
        }
        /* 彩蛋图鉴 */
        if (egg_gallery_on) {
            int gw = egg_gw / 5, gh = egg_gh / 10; if (gw < 1) gw = 1; if (gh < 1) gh = 1;
            canvas.fill(egg_gx, egg_gy, egg_gw, egg_gh, ' ');
            canvas.rect(egg_gx, egg_gy, egg_gw, egg_gh, '+');
            canvas.put_text(egg_gx + 5, egg_gy + 2, "彩蛋图鉴 (21+3)");
            int visible = std::max(1, gh - 2);
            int got = 0;
            for (int i = 0; i < 24; i++) if (egg_collected.count(i) && egg_collected[i]) got++;
            for (int i = 0; i < visible; i++) {
                int idx = egg_scroll + i;
                if (idx < 0 || idx >= 24) break;
                std::string line = std::string(EGG_NO[idx]) + " " + EGG_TITLE[idx];
                if (egg_collected.count(idx) && egg_collected[idx]) line += " [已收集]";
                canvas.put_text(egg_gx + 5, egg_gy + 12 + i * 10, line);
            }
            std::string grade = "已收集 " + std::to_string(got) + "/24";
            if (got >= 24) grade += " 图书管理员!";
            else if (got >= 18) grade += " 工匠";
            else if (got >= 10) grade += " 学徒";
            canvas.put_text(egg_gx + 5, egg_gy + egg_gh - 8, grade);
        }
        canvas.render();
    }
